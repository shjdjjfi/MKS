\section{Classical Capital-Circulation Formula}

\subsection{Simple Commodity Circulation}
\[
C_1 - M - C_2,
\qquad M=p_1 q_1=p_2 q_2,
\qquad \Delta u=u(C_2)-u(C_1)>0.
\]

\subsection{General Formula of Capital}
\[
M-C-M',
\qquad M'=M+\Delta M,
\qquad \Delta M>0.
\]

\subsection{Expanded Industrial-Capital Formula}
\[
M-C\{MP,LP\}\,\cdots P\cdots\,C'-M',
\qquad C=c+v,
\qquad C'=c+v+s.
\]

\subsection{Value Composition}
\[
W=c+v+s,
\qquad \Delta W=s,
\qquad W'=W+\Delta W.
\]

\subsection{Surplus-Value Rate}
\[
e\equiv \frac{s}{v},
\qquad s=e v,
\qquad e_t=\frac{s_t}{v_t}.
\]

\subsection{Profit Rate}
\[
r\equiv \frac{s}{c+v}=\frac{e}{\frac{c}{v}+1},
\qquad r_t=\frac{s_t}{c_t+v_t}.
\]

\subsection{Organic Composition of Capital}
\[
\omega\equiv\frac{c}{v},
\qquad r=\frac{e}{\omega+1},
\qquad \frac{\partial r}{\partial \omega}=-\frac{e}{(\omega+1)^2}<0.
\]

\section{Finite-Dimensional Multi-Feature Capital Formula}

\subsection{Feature Vector}
\[
\bm{x}_t=(x_{1t},\dots,x_{Kt})^\top\in\mathbb{R}^K,
\qquad \bm{z}_t=(1,\bm{x}_t^\top)^\top.
\]

\subsection{Capital Accumulation Function}
\[
K_{t+1}=K_t+\mathcal{A}(K_t,\bm{x}_t;\bm{\theta}),
\qquad \mathcal{A}:\mathbb{R}_+\times\mathbb{R}^K\to\mathbb{R}.
\]

\subsection{Linear Multi-Feature Formula}
\[
K_{t+1}=K_t+\alpha+\bm{\beta}^\top\bm{x}_t,
\qquad \bm{\beta}=(\beta_1,\ldots,\beta_K)^\top.
\]

\subsection{Log-Linear Formula}
\[
\log K_{t+1}=\alpha+\rho\log K_t+\bm{\beta}^\top\log\bm{x}_t,
\qquad \log\bm{x}_t=(\log x_{1t},\dots,\log x_{Kt})^\top.
\]

\subsection{Cobb-Douglas-Type Capital Formula}
\[
K_{t+1}=A K_t^{\rho}\prod_{j=1}^K x_{jt}^{\beta_j},
\qquad \sum_{j=1}^K\beta_j\lessgtr 1.
\]

\subsection{General Nonparametric Formula}
\[
K_{t+1}=K_t+g(K_t,\bm{x}_t),
\qquad g\in\mathcal{G}\subset L_2(\mathcal{X}).
\]

\section{Nonlinear Coupling Formula}

\subsection{Second-Order Expansion}
\[
\mathcal{A}(\bm{x})\approx a_0+\bm{a}^\top\bm{x}+\tfrac12\bm{x}^\top\bm{H}\bm{x}.
\]

\subsection{Interaction Matrix}
\[
\bm{H}=(h_{ij})_{i,j=1}^K,
\qquad \mathcal{A}(\bm{x})=a_0+\sum_i a_i x_i+\frac12\sum_i\sum_j h_{ij}x_ix_j.
\]

\subsection{Symmetric and Asymmetric Coupling}
\[
\bm{H}_S=\tfrac12(\bm{H}+\bm{H}^\top),
\qquad \bm{H}_A=\tfrac12(\bm{H}-\bm{H}^\top),
\qquad \bm{x}^\top\bm{H}_A\bm{x}=0.
\]

\subsection{Higher-Order Coupling}
\[
\mathcal{A}(\bm{x})=a_0+\sum_i a_i x_i+\sum_{i,j}a_{ij}x_ix_j+\sum_{i,j,\ell}a_{ij\ell}x_ix_jx_\ell.
\]

\subsection{General Polynomial Approximation}
\[
\mathcal{A}(\bm{x})\approx\sum_{|\nu|\le p} a_{\nu}\bm{x}^{\nu},
\qquad \bm{x}^{\nu}=\prod_{j=1}^K x_j^{\nu_j},
\qquad |\nu|=\sum_j\nu_j.
\]

\subsection{General Nonlinear Mapping}
\[
\mathcal{A}(\bm{x})=\Phi\!\left(\sum_{m=1}^M w_m\,\sigma(\bm{b}_m^\top\bm{x}+c_m)\right).
\]

\section{Monotonicity and Marginal Effects}

\subsection{First-Order Marginal Effects}
\[
ME_j(\bm{x})\equiv\frac{\partial \mathcal{A}(\bm{x})}{\partial x_j},
\qquad j=1,\dots,K.
\]

\subsection{Gradient Representation}
\[
\nabla\mathcal{A}(\bm{x})=
\begin{bmatrix}
\partial\mathcal{A}/\partial x_1\\
\vdots\\
\partial\mathcal{A}/\partial x_K
\end{bmatrix}.
\]

\subsection{Elasticities}
\[
\varepsilon_j(\bm{x})\equiv\frac{\partial\mathcal{A}}{\partial x_j}\frac{x_j}{\mathcal{A}},
\qquad \bm{\varepsilon}(\bm{x})=\operatorname{diag}(\bm{x})\nabla\mathcal{A}(\bm{x})/\mathcal{A}(\bm{x}).
\]

\subsection{Expected Sign Restrictions}
\[
\frac{\partial \mathcal{A}}{\partial x_j}\ge 0\ (j\in\mathcal{J}_+),
\qquad \frac{\partial \mathcal{A}}{\partial x_j}\le 0\ (j\in\mathcal{J}_-).
\]

\subsection{Non-Monotone Effects}
\[
\frac{\partial \mathcal{A}}{\partial x_j}=0 \iff x_j=x_j^{\star}(\bm{x}_{-j}),
\qquad \frac{\partial^2 \mathcal{A}}{\partial x_j^2}\bigg|_{x_j=x_j^{\star}}\neq 0.
\]

\section{Second-Order Properties}

\subsection{Hessian Matrix}
\[
\nabla^2\mathcal{A}(\bm{x})=\left[\frac{\partial^2\mathcal{A}}{\partial x_i\partial x_j}\right]_{i,j=1}^K.
\]

\subsection{Concavity and Convexity}
\[
\nabla^2\mathcal{A}(\bm{x})\preceq 0 \Rightarrow \mathcal{A}\text{ concave},
\qquad
\nabla^2\mathcal{A}(\bm{x})\succeq 0 \Rightarrow \mathcal{A}\text{ convex}.
\]

\subsection{Diminishing Marginal Returns}
\[
\frac{\partial^2\mathcal{A}}{\partial x_j^2}<0,
\qquad \frac{\partial ME_j}{\partial x_j}<0.
\]

\subsection{Increasing Marginal Returns}
\[
\frac{\partial^2\mathcal{A}}{\partial x_j^2}>0,
\qquad \frac{\partial ME_j}{\partial x_j}>0.
\]

\subsection{Complementarity and Substitution}
\[
\frac{\partial^2\mathcal{A}}{\partial x_i\partial x_j}>0\Rightarrow \text{complementarity},
\qquad
\frac{\partial^2\mathcal{A}}{\partial x_i\partial x_j}<0\Rightarrow \text{substitution}.
\]

\subsection{Cross-Partial Effects}
\[
CPE_{ij}(\bm{x})\equiv\frac{\partial^2\mathcal{A}(\bm{x})}{\partial x_i\partial x_j},
\qquad CPE_{ij}=CPE_{ji}\ \text{if }\mathcal{A}\in C^2.
\]

\section{Bounds and Constraints}

\subsection{Lower Bounds}
\[
K_t\ge \underline{K}>0,
\qquad x_{jt}\ge \underline{x}_j.
\]

\subsection{Positive Accumulation Condition}
\[
\mathcal{A}(K_t,\bm{x}_t)\ge 0,
\qquad K_{t+1}-K_t\ge 0.
\]

\subsection{Upper Bounds}
\[
K_t\le \overline{K},
\qquad x_{jt}\le \overline{x}_j,
\qquad \mathcal{A}(K_t,\bm{x}_t)\le \overline{A}.
\]

\subsection{Resource Constraints}
\[
\bm{p}_t^\top\bm{x}_t\le B_t,
\qquad \sum_{j=1}^K a_{rj}x_{jt}\le R_{rt},\ \ r=1,\dots,R.
\]

\subsection{Ecological Constraints}
\[
E_t(\bm{x}_t)\le \overline{E}_t,
\qquad E_t(\bm{x}_t)=\sum_{j=1}^K \eta_j x_{jt}+\sum_{i,j}\eta_{ij}x_{it}x_{jt}.
\]

\subsection{Market Constraints}
\[
D_t(P_t)\ge Q_t(\bm{x}_t),
\qquad \Pi_t=P_tQ_t-C_t(\bm{x}_t)\ge 0.
\]

\subsection{Feasible Set}
\[
\mathcal{F}_t=\{\bm{x}\in\mathbb{R}^K_+: \bm{p}_t^\top\bm{x}\le B_t,\ E_t(\bm{x})\le\overline{E}_t,\ D_t(P_t)\ge Q_t(\bm{x})\}.
\]

\section{Optimization and Critical Points}

\subsection{Unconstrained Maximization}
\[
\max_{\bm{x}\in\mathbb{R}^K} \ \mathcal{A}(\bm{x}),
\qquad \nabla\mathcal{A}(\bm{x}^{\star})=\bm{0}.
\]

\subsection{Constrained Maximization}
\[
\max_{\bm{x}\in\mathcal{F}}\ \mathcal{A}(\bm{x}),
\qquad g_m(\bm{x})\le 0,\ m=1,\dots,M.
\]

\subsection{Lagrangian Formulation}
\[
\mathcal{L}(\bm{x},\bm{\lambda})=\mathcal{A}(\bm{x})-\sum_{m=1}^M\lambda_m g_m(\bm{x}),
\qquad \bm{\lambda}\ge\bm{0}.
\]

\subsection{Karush-Kuhn-Tucker Conditions}
\[
\nabla_{\bm{x}}\mathcal{L}(\bm{x}^{\star},\bm{\lambda}^{\star})=\bm{0},
\quad g_m(\bm{x}^{\star})\le 0,
\quad \lambda_m^{\star}\ge 0,
\quad \lambda_m^{\star}g_m(\bm{x}^{\star})=0.
\]

\subsection{Budget-Constrained Capital Allocation}
\[
\max_{\bm{x}\ge\bm{0}}\ \mathcal{A}(\bm{x})
\quad\text{s.t.}\quad \bm{p}^\top\bm{x}\le B,
\qquad \frac{\partial\mathcal{A}/\partial x_j}{p_j}=\lambda\ \text{if }x_j>0.
\]

\subsection{Ecological-Constrained Capital Allocation}
\[
\max_{\bm{x}\ge\bm{0}}\ \mathcal{A}(\bm{x})
\quad\text{s.t.}\quad E(\bm{x})\le \overline{E},
\qquad \nabla\mathcal{A}(\bm{x}^{\star})=\mu\nabla E(\bm{x}^{\star})+\nu\bm{p}.
\]

\subsection{Classification of Critical Points}
\[
\nabla\mathcal{A}(\bm{x}^{\star})=\bm{0},
\qquad \nabla^2\mathcal{A}(\bm{x}^{\star})\prec 0\Rightarrow \text{strict local max},
\qquad \nabla^2\mathcal{A}(\bm{x}^{\star})\succ 0\Rightarrow \text{strict local min}.
\]

\section{Thresholds and Regime Shifts}

\subsection{Single-Threshold Formula}
\[
K_{t+1}=K_t+\alpha_1+\bm{\beta}_1^\top\bm{x}_t+\left(\alpha_2+\bm{\beta}_2^\top\bm{x}_t\right)\mathbf{1}\{q_t>\tau\}.
\]

\subsection{Financial-Leverage Threshold}
\[
q_t\equiv \ell_t=\frac{D_t}{A_t},
\qquad \mathcal{A}_t=\mathcal{A}_L(\bm{x}_t)\mathbf{1}\{\ell_t\le\tau_\ell\}+\mathcal{A}_H(\bm{x}_t)\mathbf{1}\{\ell_t>\tau_\ell\}.
\]

\subsection{Platform-Network Threshold}
\[
q_t\equiv n_t,
\qquad \mathcal{A}_t=\gamma_0+\gamma_1 n_t+\gamma_2 n_t^2\mathbf{1}\{n_t>\tau_n\}.
\]

\subsection{Ecological-Constraint Threshold}
\[
q_t\equiv E_t,
\qquad \mathcal{A}_t=\delta_0+\delta_1 E_t+\delta_2(E_t-\tau_E)_+.
\]

\subsection{Structural Break}
\[
Y_t=\bm{x}_t^\top\bm{\beta}_1\mathbf{1}\{t\le T_b\}+\bm{x}_t^\top\bm{\beta}_2\mathbf{1}\{t>T_b\}+u_t.
\]

\subsection{Regime-Switching Formula}
\[
K_{t+1}=K_t+\alpha_{S_t}+\bm{\beta}_{S_t}^\top\bm{x}_t+\varepsilon_t,
\qquad \Pr(S_t=j\mid S_{t-1}=i)=p_{ij}.
\]

\section{Dynamic Capital Accumulation}

\subsection{Discrete-Time Dynamics}
\[
K_{t+1}=F(K_t,\bm{x}_t),
\qquad F(K_t,\bm{x}_t)=K_t+\mathcal{A}(K_t,\bm{x}_t).
\]

\subsection{Recursive Growth}
\[
K_t=F^{(t)}(K_0;\{\bm{x}_s\}_{s=0}^{t-1}),
\qquad K_{t+h}=F^{(h)}(K_t;\{\bm{x}_{t+s}\}_{s=0}^{h-1}).
\]

\subsection{Growth Rate Representation}
\[
g_t\equiv\frac{K_{t+1}-K_t}{K_t}=\frac{\mathcal{A}(K_t,\bm{x}_t)}{K_t},
\qquad \log\frac{K_{t+1}}{K_t}=\log(1+g_t).
\]

\subsection{Steady State}
\[
K^{\star}=F(K^{\star},\bm{x}^{\star}),
\qquad \mathcal{A}(K^{\star},\bm{x}^{\star})=0.
\]

\subsection{Stability Condition}
\[
\left|\frac{\partial F}{\partial K}(K^{\star},\bm{x}^{\star})\right|<1,
\qquad \rho\!\left(\nabla_{\bm{s}}\bm{F}(\bm{s}^{\star})\right)<1.
\]

\subsection{Continuous-Time Dynamics}
\[
\dot K(t)=\mathcal{A}(K(t),\bm{x}(t)),
\qquad K(t)=K(0)+\int_0^t\mathcal{A}(K(u),\bm{x}(u))\,du.
\]

\subsection{Crisis and Collapse Conditions}
\[
\mathcal{A}(K_t,\bm{x}_t)<0\ \forall t\in[t_0,t_1],
\qquad K_{t_1}<\underline{K},
\qquad \exists t:\ \frac{\partial F}{\partial K}>1\ \text{and}\ \mathcal{A}<0.
\]

\section{Econometric Specification}

\subsection{Panel Data Model}
\[
K_{i,t+1}=K_{it}+\mathcal{A}(K_{it},\bm{x}_{it})+u_{it},
\qquad i=1,\dots,N,\ t=1,\dots,T.
\]

\subsection{Linear Fixed-Effects Model}
\[
K_{i,t+1}=\alpha_i+\tau_t+\rho K_{it}+\bm{\beta}^\top\bm{x}_{it}+u_{it}.
\]

\subsection{Interaction Fixed-Effects Model}
\[
K_{i,t+1}=\alpha_i+\tau_t+\rho K_{it}+\bm{\beta}^\top\bm{x}_{it}+\sum_{j<\ell}\gamma_{j\ell}x_{jit}x_{\ell it}+u_{it}.
\]

\subsection{Dynamic Panel Model}
\[
K_{i,t+1}=\alpha_i+\tau_t+\phi K_{it}+\psi K_{i,t-1}+\bm{\beta}^\top\bm{x}_{it}+u_{it}.
\]

\subsection{Threshold Regression Model}
\[
K_{i,t+1}=\alpha_i+\tau_t+\bm{x}_{it}^\top\bm{\beta}_1\mathbf{1}\{q_{it}\le\tau\}+\bm{x}_{it}^\top\bm{\beta}_2\mathbf{1}\{q_{it}>\tau\}+u_{it}.
\]

\subsection{Nonparametric Regression Model}
\[
K_{i,t+1}=m(K_{it},\bm{x}_{it})+u_{it},
\qquad \mathbb{E}[u_{it}\mid K_{it},\bm{x}_{it}]=0.
\]

\subsection{Error Decomposition}
\[
u_{it}=\mu_i+\lambda_t+\varepsilon_{it},
\qquad u_{it}=\nu_{it}+\xi_{it},
\qquad \mathbb{E}[\varepsilon_{it}\varepsilon_{js}]=\sigma_{ij,ts}.
\]

\section{Identification and Endogeneity}

\subsection{Exogeneity Condition}
\[
\mathbb{E}[u_{it}\mid K_{it},\bm{x}_{it},\alpha_i,\tau_t]=0.
\]

\subsection{Endogeneity}
\[
\operatorname{Cov}(x_{jit},u_{it})\neq 0,
\qquad \mathbb{E}[u_{it}\mid \bm{x}_{it}]\neq 0.
\]

\subsection{Reverse Causality}
\[
x_{jit}=\pi_0+\pi_1 K_{i,t+1}+v_{jit},
\qquad \operatorname{Cov}(v_{jit},u_{it})\neq 0.
\]

\subsection{Omitted Variable Bias}
\[
K_{i,t+1}=\bm{x}_{it}^\top\bm{\beta}+\delta w_{it}+u_{it},
\qquad \tilde{\bm{\beta}}=\bm{\beta}+\delta\left(\mathbb{E}[\bm{x}\bm{x}^\top]\right)^{-1}\mathbb{E}[\bm{x}w].
\]

\subsection{Instrumental Variables}
\[
\mathbb{E}[\bm{z}_{it}u_{it}]=\bm{0},
\qquad \operatorname{rank}(\mathbb{E}[\bm{z}_{it}\bm{x}_{it}^\top])=K,
\qquad \hat{\bm{\beta}}_{2SLS}=(X^\top P_Z X)^{-1}X^\top P_Z Y.
\]

\subsection{Difference-in-Differences Form}
\[
Y_{it}=\alpha_i+\tau_t+\theta(D_i\times Post_t)+\bm{w}_{it}^\top\bm{\gamma}+u_{it},
\qquad \theta=\Delta\bar Y_T-\Delta\bar Y_C.
\]

\subsection{Regression Discontinuity Form}
\[
Y_i=\alpha+\tau\mathbf{1}\{r_i\ge c\}+f_-(r_i-c)\mathbf{1}\{r_i<c\}+f_+(r_i-c)\mathbf{1}\{r_i\ge c\}+u_i,
\qquad \lim_{r\downarrow c}\mathbb{E}[Y_i(0)\mid r_i=r]=\lim_{r\uparrow c}\mathbb{E}[Y_i(0)\mid r_i=r].
\]

\subsection{Moment Conditions}
\[
\mathbb{E}[\bm{g}(W_i,\bm{\theta}_0)]=\bm{0},
\qquad \bm{g}(W_i,\bm{\theta})=\bm{z}_i\big(Y_i-h(X_i;\bm{\theta})\big).
\]

\section{Infinite-Dimensional Capital Formula}

\subsection{Measure Space}
\[
(\Omega,\mathcal{F},\mu),
\qquad x_t:\Omega\to\mathbb{R},
\qquad x_t\in L_2(\Omega,\mu).
\]

\subsection{Feature Function}
\[
\varphi_t(\omega)=x_t(\omega),
\qquad \bm{\varphi}_t=(\varphi_{1t},\varphi_{2t},\ldots).
\]

\subsection{Hilbert Space Representation}
\[
\mathcal{H}=L_2(\Omega,\mu),
\qquad \langle f,g\rangle_{\mathcal{H}}=\int_\Omega f(\omega)g(\omega)d\mu(\omega),
\qquad \|f\|_{\mathcal{H}}^2=\langle f,f\rangle.
\]

\subsection{Linear Infinite-Dimensional Formula}
\[
K_{t+1}=K_t+a_0+\langle b, x_t\rangle_{\mathcal{H}},
\qquad b\in\mathcal{H}.
\]

\subsection{Quadratic Kernel Formula}
\[
K_{t+1}=K_t+a_0+\langle b,x_t\rangle+\iint_{\Omega\times\Omega}x_t(\omega)\,\kappa(\omega,\omega')\,x_t(\omega')\,d\mu(\omega)d\mu(\omega').
\]

\subsection{Integral Operator}
\[
(\mathcal{K}f)(\omega)=\int_\Omega \kappa(\omega,\omega')f(\omega')d\mu(\omega'),
\qquad \mathcal{K}:\mathcal{H}\to\mathcal{H}.
\]

\subsection{Kernel Symmetry}
\[
\kappa(\omega,\omega')=\kappa(\omega',\omega),
\qquad \langle f,\mathcal{K}g\rangle=\langle \mathcal{K}f,g\rangle.
\]

\subsection{Kernel Boundedness}
\[
\sup_{\|f\|\le 1}\|\mathcal{K}f\|<\infty,
\qquad \iint |\kappa(\omega,\omega')|^2d\mu(\omega)d\mu(\omega')<\infty.
\]

\subsection{Existence of Integrals}
\[
\int_\Omega |b(\omega)x_t(\omega)|d\mu(\omega)<\infty,
\qquad \iint|x_t(\omega)\kappa(\omega,\omega')x_t(\omega')|d\mu d\mu<\infty.
\]

\subsection{Infinite-Dimensional Marginal Effect}
\[
\frac{\delta\mathcal{A}[x]}{\delta x(\omega)}=b(\omega)+2\int_\Omega \kappa(\omega,\omega')x(\omega')d\mu(\omega').
\]

\subsection{Infinite-Dimensional Critical Point}
\[
\frac{\delta\mathcal{A}[x^{\star}]}{\delta x(\omega)}=0\ \text{a.e. }\omega,
\qquad b+2\mathcal{K}x^{\star}=0,
\qquad x^{\star}=-\tfrac12\mathcal{K}^{-1}b.
\]

\section{Finite-Dimensional Approximation}

\subsection{Basis Expansion}
\[
x(\omega)=\sum_{m=1}^{\infty} \xi_m\phi_m(\omega),
\qquad \langle\phi_m,\phi_n\rangle=\mathbf{1}\{m=n\}.
\]

\subsection{Orthogonal Projection}
\[
\Pi_M x=\sum_{m=1}^M\xi_m\phi_m,
\qquad \xi_m=\langle x,\phi_m\rangle.
\]

\subsection{Truncated Approximation}
\[
K_{t+1}\approx K_t+a_0+\sum_{m=1}^M b_m\xi_{mt}+\sum_{m=1}^M\sum_{n=1}^M\kappa_{mn}\xi_{mt}\xi_{nt}.
\]

\subsection{Approximation Error}
\[
R_M(x)=x-\Pi_Mx,
\qquad \|R_M(x)\|_{\mathcal{H}}^2=\sum_{m>M}\xi_m^2\to 0.
\]

\subsection{Sieve Approximation}
\[
g(\bm{x})\approx g_M(\bm{x})=\sum_{m=1}^M \theta_m\psi_m(\bm{x}),
\qquad \mathcal{S}_M=\operatorname{span}\{\psi_1,\dots,\psi_M\}.
\]

\subsection{Sieve Estimation}
\[
\hat{\bm{\theta}}_M=\arg\min_{\bm{\theta}\in\mathbb{R}^M}\sum_{i,t}\left(K_{i,t+1}-K_{it}-\sum_{m=1}^M\theta_m\psi_m(\bm{x}_{it})\right)^2.
\]

\subsection{Convergence Conditions}
\[
M\to\infty,
\qquad \frac{M}{n}\to 0,
\qquad \|\hat g_M-g_0\|_{L_2}\xrightarrow{p}0.
\]

\section{Statistical Estimation}

\subsection{Least Squares Estimator}
\[
\hat{\bm{\beta}}_{OLS}=\arg\min_{\bm{\beta}}\|\bm{Y}-X\bm{\beta}\|_2^2=(X^\top X)^{-1}X^\top\bm{Y}.
\]

\subsection{Regularized Estimator}
\[
\hat{\bm{\beta}}_{\lambda}=\arg\min_{\bm{\beta}}\left\{\frac{1}{n}\|\bm{Y}-X\bm{\beta}\|_2^2+\lambda\|\bm{\beta}\|_2^2\right\}.
\]

\subsection{Penalized High-Dimensional Estimator}
\[
\hat{\bm{\beta}}_{LASSO}=\arg\min_{\bm{\beta}}\left\{\frac{1}{2n}\|\bm{Y}-X\bm{\beta}\|_2^2+\lambda\|\bm{\beta}\|_1\right\}.
\]

\subsection{Kernel Estimator}
\[
\hat m(\bm{x})=\frac{\sum_{i=1}^n K_h(\bm{X}_i-\bm{x})Y_i}{\sum_{i=1}^n K_h(\bm{X}_i-\bm{x})},
\qquad K_h(\bm{u})=h^{-d}K(\bm{u}/h).
\]

\subsection{Generalized Method of Moments}
\[
\hat{\bm{\theta}}_{GMM}=\arg\min_{\bm{\theta}}\ \bar{\bm{g}}_n(\bm{\theta})^\top W_n\bar{\bm{g}}_n(\bm{\theta}),
\qquad \bar{\bm{g}}_n(\bm{\theta})=\frac1n\sum_{i=1}^n\bm{g}(W_i,\bm{\theta}).
\]

\subsection{Asymptotic Normality}
\[
\sqrt{n}(\hat{\bm{\theta}}-\bm{\theta}_0)\xrightarrow{d}\mathcal{N}(\bm{0},V),
\qquad V=(G^\top WG)^{-1}G^\top W\Omega WG(G^\top WG)^{-1}.
\]

\subsection{Variance Estimation}
\[
\hat V_{HC}=(X^\top X)^{-1}\left(\sum_{i=1}^n \hat u_i^2\bm{x}_i\bm{x}_i^\top\right)(X^\top X)^{-1},
\qquad \hat V_{cluster}= (X^\top X)^{-1}\left(\sum_{g=1}^G X_g^\top\hat u_g\hat u_g^\top X_g\right)(X^\top X)^{-1}.
\]

\section{Hypothesis Testing}

\subsection{Linearity Test}
\[
H_0:\gamma_{j\ell}=0\ \forall j<\ell,
\qquad F=\frac{(SSR_r-SSR_u)/q}{SSR_u/(n-k_u)}.
\]

\subsection{Coupling Test}
\[
H_0:\bm{H}=\bm{0},
\qquad W=n\,\hat{\bm{h}}^\top\left[\widehat{\operatorname{Var}}(\hat{\bm{h}})\right]^{-1}\hat{\bm{h}}\xrightarrow{d}\chi_q^2.
\]

\subsection{Threshold Test}
\[
H_0:\bm{\beta}_1=\bm{\beta}_2,
\qquad LR(\tau)=\frac{SSR_0-SSR_1(\tau)}{\hat\sigma^2},
\qquad \sup_{\tau\in\mathcal{T}}LR(\tau).
\]

\subsection{Monotonicity Test}
\[
H_0:\frac{\partial\mathcal{A}}{\partial x_j}\ge 0\ \forall \bm{x}\in\mathcal{X},
\qquad T_n=\inf_{\bm{x}\in\mathcal{X}}\frac{\widehat{\partial\mathcal{A}/\partial x_j}(\bm{x})}{\hat s_j(\bm{x})}.
\]

\subsection{Infinite-Dimensional Relevance Test}
\[
H_0:b(\omega)=0\ \text{a.e.},
\qquad T_b=n\sum_{m=1}^{M_n}\frac{\hat b_m^2}{\hat\sigma_m^2}.
\]

\subsection{Kernel Interaction Test}
\[
